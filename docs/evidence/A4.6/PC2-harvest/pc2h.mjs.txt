const ev = async (page, fn, arg) => { for (let i = 0; i < 6; i++) { try { return await page.evaluate(fn, arg); } catch (e) { if (!/context was destroyed|navigation/i.test(e.message)) throw e; await page.waitForTimeout(800); } } return page.evaluate(fn, arg); };

const picto = (id) => `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="52" r="30" fill="#f4b860" stroke="#000" stroke-width="5"/><circle cx="40" cy="46" r="4" fill="#000"/><circle cx="60" cy="46" r="4" fill="#000"/><path d="M38 62 Q50 72 62 62" fill="none" stroke="#000" stroke-width="4"/><text x="50" y="96" font-size="11" text-anchor="middle" fill="#000">${id}</text></svg>`;

// normalised outerHTML: ids React / Radix generate per mount, srcs and inline styles are dropped
const OUTER = (sel) => [...document.querySelectorAll(sel)].map((el) => { const c = el.cloneNode(true); for (const n of [c, ...c.querySelectorAll('*')]) for (const a of [...n.attributes]) if (/^(id|for|src|srcset|style|aria-controls|aria-labelledby|aria-describedby|data-radix-.*)$/.test(a.name) || /radix-/.test(a.value)) n.removeAttribute(a.name); return c.outerHTML; }).join('\n');

async function run(s, w, h, dark) {
  const ctx = await browser.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 1, colorScheme: dark ? 'dark' : 'light' });
  await ctx.addInitScript(INIT, [s.role, s.student || 's-0001', (s.id.startsWith('board') && opt.style) || s.boardStyle || '', dark, LEGACY ? [] : s.ivory, s.calView || '']);
  await ctx.route((u) => !/^(http:\/\/127\.0\.0\.1:\d+|data:|about:|blob:)/.test(String(u)), (route) => {
    const u = route.request().url();
    if (/static\.arasaac\.org|api\.arasaac\.org\/v1\/pictograms\/\d+/.test(u)) { const id = (u.match(/(\d{3,})/) || [])[1] || '0'; return route.fulfill({ contentType: 'image/svg+xml', body: picto(id) }); }
    if (/api\.arasaac\.org\/v1\/pictograms\/en\/(best)?search/.test(u)) return route.fulfill({ contentType: 'application/json', body: JSON.stringify([2349, 2350, 2351, 2352, 2353, 2354].map((id, i) => ({ _id: id, keywords: [{ keyword: ['cat', 'kitten', 'pet cat', 'tabby cat', 'cat face', 'cats'][i] }] }))) });
    return route.abort();
  });
  const page = await ctx.newPage();
  await page.clock.setFixedTime(new Date('2026-11-13T12:00:00'));
  const errors = []; page.on('pageerror', (e) => errors.push(String(e.message).slice(0, 160)));
  await page.goto(BASE + s.route, { waitUntil: 'load', timeout: 120000 }).catch((e) => errors.push('goto ' + e.message.slice(0, 100)));
  const t0 = Date.now();
  while (Date.now() - t0 < 40000) { const st = await ev(page, () => ({ now: performance.now(), act: Math.max(window.__lastResEnd || 0, window.__lastRequestAt || 0), txt: (document.getElementById('root')?.innerText || '').length })); if (st.now - st.act > 1000 && st.txt > 100) break; await page.waitForTimeout(250); }
  await page.waitForTimeout(2600);
  const m = { errors, attr: await ev(page, () => document.documentElement.getAttribute('data-hec-theme')) };
  if (s.action) { try { await s.action(page); await page.waitForTimeout(1200); } catch (e) { m.actionError = String(e.message).slice(0, 200); } }
  if (s.probe) { try { m.probe = await ev(page, s.probe); } catch (e) { m.probeError = String(e.message).slice(0, 200); } }
  m.overflow = await ev(page, () => document.documentElement.scrollWidth > document.documentElement.clientWidth);
  if (LEGACY) {
    m.dom = await ev(page, OUTER, s.dom);
  } else {
    const p = path.join(outDir, `${label}-${s.id}-${w}-${dark ? 'dark' : 'light'}.png`);
    await page.screenshot({ path: p });
    m.shot = path.basename(p);
  }
  await ctx.close();
  return m;
}
