import React from "react";
import { createRoot } from "react-dom/client";
import App from "@/App.tsx";
import "@/index.css";
import "react-quill-new/dist/quill.snow.css";
import { IVORY_ROUTES } from "@/lib/ivoryScope";

// PC2-harvest harness only: the routes to show in the new design are listed HERE, at run time, from
// localStorage pc2h_ivory (a JSON array of paths). src/lib/ivoryScope.ts is never edited.
try {
  const extra = JSON.parse(localStorage.getItem("pc2h_ivory") || "[]");
  for (const path of extra) IVORY_ROUTES.push({ path });
} catch {}

createRoot(document.getElementById("root")!).render(<App />);
