// Review probe: renders editor screens from the r21 harness (fake client, made-up data) at 375 light/dark,
// checks toolbar keyboard roving past disabled buttons, and an old row holding a (made-up) base64 picture.
//   node review-run.mjs <repoRoot> <outDir>
import fs from 'node:fs'; import path from 'node:path'; import { spawn } from 'node:child_process';
const H = 'C:/Users/Dell/hec-tools/r21-harness';
const { chromium } = await import('file:///' + H + '/node_modules/playwright/index.mjs');
const repo = path.resolve(process.argv[2]); const outDir = path.resolve(process.argv[3]);
fs.mkdirSync(outDir, { recursive: true });
const PORT = 5437, BASE = `http://127.0.0.1:${PORT}`;

const pv = path.join(repo, '.claude/review-preview'); fs.mkdirSync(pv, { recursive: true });
for (const f of fs.readdirSync(path.join(H, 'preview-editor'))) fs.copyFileSync(path.join(H, 'preview-editor', f), path.join(pv, f));
// Made-up picture row: homework 2 gets a 1x1 base64 PNG between two paragraphs (in this copy only).
let fake = fs.readFileSync(path.join(pv, 'fake-supabase.ts'), 'utf8');
const PIC = '<p>Look at the picture:</p><p><img src="data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mP8z8BQDwAEhQGAhKmMIQAAAABJRU5ErkJggg=="></p><p>Write three sentences.</p>';
fake = fake.replace(/(db\.homeworks\[1\]\.body = [^\n]*\n)/, `$1db.homeworks[1].body = ${JSON.stringify(PIC)};\n`);
fs.writeFileSync(path.join(pv, 'fake-supabase.ts'), fake);
await new Promise((r) => spawn(process.execPath, [path.join(H, 'gen-schema.mjs'), repo, path.join(pv, 'schema.json')], { stdio: 'inherit' }).on('exit', r));
const vite = spawn(process.execPath, [path.join(repo, 'node_modules/vite/bin/vite.js'), '--config', path.join(pv, 'vite.config.ts'), '--port', String(PORT), '--strictPort'], { cwd: repo, stdio: 'ignore' });
process.on('exit', () => { try { vite.kill(); } catch {} });
for (let i = 0; i < 120; i++) { try { if ((await fetch(BASE + '/')).ok) break; } catch {} await new Promise((r) => setTimeout(r, 1000)); }
const browser = await chromium.launch({ channel: 'chrome' });

const INIT = ([role, student, dark, fakeHw]) => {
  try {
    localStorage.setItem('fake_role', role);
    if (role === 'student') { localStorage.setItem('selected_student_id', student); localStorage.setItem('fake_student', student); }
    if (fakeHw) localStorage.setItem('fake_hw', '1'); else localStorage.removeItem('fake_hw');
    for (const u of ['u-admin', 'u-teacher', 'u-student', 'u-family']) localStorage.setItem('hec-theme:' + u, dark ? 'dark' : 'light');
    localStorage.setItem('hec-theme-boot', dark ? 'dark' : 'light');
  } catch {}
  const apply = () => { const de = document.documentElement; if (!de) return false; if (dark) de.classList.add('dark'); return true; };
  if (!apply()) new MutationObserver((_, o) => { if (apply()) o.disconnect(); }).observe(document, { childList: true });
};

async function open(route, role, dark, fakeHw, w = 375, h = 812) {
  const ctx = await browser.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 1, colorScheme: dark ? 'dark' : 'light', hasTouch: w < 768 });
  await ctx.addInitScript(INIT, [role, 's-0001', dark, fakeHw]);
  await ctx.route((u) => !/^(http:\/\/127\.0\.0\.1:\d+|data:|about:|blob:)/.test(String(u)), (route) => route.abort());
  const page = await ctx.newPage();
  const errors = []; page.on('pageerror', (e) => errors.push(String(e.message).slice(0, 200)));
  await page.goto(BASE + route, { waitUntil: 'load', timeout: 300000 });
  await page.waitForTimeout(5000);
  return { ctx, page, errors };
}

const results = {};
// warm-up compile
{ const { ctx } = await open('/teacher/assignments', 'teacher', false, false, 1280, 800); await ctx.close(); }

for (const dark of [false, true]) {
  const t = dark ? 'dark' : 'light';
  // 1. AssignmentUpload (full preset), new homework
  {
    const { ctx, page, errors } = await open('/teacher/assignments', 'teacher', dark, false);
    const root = page.locator('.rte').first();
    await root.waitFor({ state: 'visible', timeout: 30000 }); await root.scrollIntoViewIfNeeded(); await page.waitForTimeout(500);
    await page.screenshot({ path: path.join(outDir, `assign-375-${t}.png`) });
    // roving: walk ArrowRight through the whole toolbar from the first button, record each focused label
    const first = root.locator('.rte-toolbar button').first();
    await first.focus();
    const seen = [];
    for (let i = 0; i < 30; i++) { seen.push(await page.evaluate(() => document.activeElement?.getAttribute('aria-label') || document.activeElement?.tagName)); await page.keyboard.press('ArrowRight'); }
    const disabled = await root.evaluate((r) => [...r.querySelectorAll('.rte-toolbar button')].filter((b) => b.disabled).map((b) => b.getAttribute('aria-label')));
    const all = await root.evaluate((r) => [...r.querySelectorAll('.rte-toolbar button')].map((b) => b.getAttribute('aria-label')));
    const small = await root.evaluate((r) => [...r.querySelectorAll('button, input')].filter((b) => { const x = b.getBoundingClientRect(); return x.width > 0 && (x.width < 43.5 || x.height < 43.5); }).length);
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
    // Tab stop after focusing a button that then becomes disabled: Outdent in a paragraph at indent 1 -> 0
    const ce = root.locator('[contenteditable=true]');
    await ce.click(); await page.keyboard.type('Hello');
    await root.getByRole('button', { name: 'Indent' }).click();
    const outdent = root.getByRole('button', { name: 'Outdent' });
    await outdent.focus(); await page.keyboard.press('Enter');
    await page.waitForTimeout(200);
    const afterOutdent = await root.evaluate((r) => ({
      active: document.activeElement?.getAttribute('aria-label') || document.activeElement?.tagName,
      tabStops: [...r.querySelectorAll('.rte-toolbar button')].filter((b) => b.tabIndex === 0).map((b) => ({ label: b.getAttribute('aria-label'), disabled: b.disabled })),
    }));
    results[`assign-${t}`] = { seen, all, disabled, small, overflow, afterOutdent, errors };
    await ctx.close();
  }
  // 2. Student answer box (minimal preset) in the homework dialog
  {
    const { ctx, page, errors } = await open('/student/assignments', 'student', dark, true);
    await page.getByText('Unit 3 - Colours worksheet').first().click(); await page.waitForTimeout(3000);
    const root = page.locator('[role=dialog] .rte').first();
    await root.waitFor({ state: 'visible', timeout: 30000 }); await root.scrollIntoViewIfNeeded(); await page.waitForTimeout(500);
    const ce = root.locator('[contenteditable=true]');
    await ce.click(); await page.keyboard.type('My name is Minh. ');
    await page.screenshot({ path: path.join(outDir, `student-hw-375-${t}.png`) });
    const fontSize = await ce.evaluate((el) => getComputedStyle(el).fontSize);
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
    results[`student-${t}`] = { fontSize, overflow, errors };
    await ctx.close();
  }
}
// 3. Old row with a picture, opened in EditHomeworkDialog: what the teacher sees, and what an edit saves.
{
  const { ctx, page, errors } = await open('/teacher/assignments', 'teacher', false, false, 1280, 900);
  await page.addInitScript(() => {});
  const edits = page.locator('main').getByRole('button', { name: /^Edit$/ });
  const n = await edits.count();
  let found = null;
  for (let i = 0; i < n && !found; i++) {
    await edits.nth(i).click(); await page.waitForTimeout(2500);
    const root = page.locator('[role=dialog] .rte').first();
    if (await root.count()) {
      const html = await root.locator('[contenteditable=true]').evaluate((el) => el.innerHTML);
      if (html.includes('Look at the picture')) found = { i, html };
      else { await page.keyboard.press('Escape'); await page.waitForTimeout(800); }
    }
  }
  if (found) {
    await page.screenshot({ path: path.join(outDir, `edit-picture-row-1280-light.png`) });
    await page.evaluate(() => { window.__payloads = []; });
    const ce = page.locator('[role=dialog] .rte [contenteditable=true]');
    await ce.click(); await page.keyboard.press('Control+End'); await page.keyboard.type(' Thanks.');
    const save = page.locator('[role=dialog]').getByRole('button', { name: /^(Save Changes|Save changes|Save|Update)/ }).first();
    await save.click(); await page.waitForTimeout(2500);
    const payloads = await page.evaluate(() => (window.__payloads || []).map((p) => ({ table: p.table, method: p.method, body: (Array.isArray(p.payload) ? p.payload[0] : p.payload || {}).body })));
    results.pictureRow = { editorHtml: found.html, payloads, errors };
  } else results.pictureRow = { notFound: true, n, errors };
  await ctx.close();
}
fs.writeFileSync(path.join(outDir, 'review-results.json'), JSON.stringify(results, null, 1));
console.log(JSON.stringify(results, null, 1));

process.exit(0);
